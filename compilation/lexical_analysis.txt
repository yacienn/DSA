Algorithm 
Begin
    Read(token)
    TC ← first character of token
    EC ← C1                       // initial state

    While (EC <> ∅ and TC <> #) do
        EC ← T[EC, TC]
        TC ← next character of token
    End while

    If (EC = ∅) then
        Write("lexical error")
    Else if (EC <> C4) then                   // EC is not the final state
        Write("lexical error")
    Else
        Write("Correct token")
        Encode(token)
        Insert(token)
    End if
End
